package org.eclipse.swt.m3;

import java.util.*;

/* steady-state cost of value updates of 100 rows in a tree-mode filter (random recursive forest); median of the last 200 of 300 */
public final class ProbeTree {
	public static void main (String [] args) {
		int n = Integer.parseInt (args [0]);
		Random random = new Random (1);
		String [] texts = new String [n];
		int [] parents = new int [n];
		for (int i = 0; i < n; i++) {
			texts [i] = "w" + random.nextInt (5000);
			parents [i] = i == 0 || random.nextInt (8) == 0 ? -1 : random.nextInt (i);
		}
		M3ColumnFilter filter = new M3ColumnFilter (1, n, (row, column) -> texts [row]);
		filter.setParents (row -> parents [row]);
		filter.setText (0, "w12", M3TextMatch.CONTAINS);
		int rounds = 300;
		long [] t = new long [rounds];
		int [] rows = new int [100];
		for (int round = 0; round < rounds; round++) {
			for (int k = 0; k < 100; k++) {
				rows [k] = random.nextInt (n);
				texts [rows [k]] = "w" + random.nextInt (5000);
			}
			long start = System.nanoTime ();
			filter.rowsChanged (rows);
			t [round] = System.nanoTime () - start;
		}
		Arrays.sort (t, 100, rounds);
		System.err.println ("tree " + n + " change100 " + t [200] / 1000 + " us (size " + filter.size () + ")");
	}
}
