import java.util.*;
import java.util.logging.*;

import org.eclipse.swt.*;
import org.eclipse.swt.graphics.*;
import org.eclipse.swt.snippets.m3.*;
import org.eclipse.swt.widgets.*;

/**
 * Screenshot and state harness for the Tree examples (examples/.../snippets/m3/M3Tree*.java), run on the invisible
 * desktop by tree_example_shots.sh: opens each example, drives it to a few states, logs what a client reads
 * (JUL, one line per state, comparable across the default mode, the original mode and the official jar) and, with
 * a directory argument, saves the shell of every state as PNG (Control.print). Native keys are sent as messages,
 * as the tests do, because the invisible desktop gets no real input.
 */
public class TreeExampleShots {
	static {
		/* one plain line per record, set before the first logger initialises SimpleFormatter */
		System.setProperty ("java.util.logging.SimpleFormatter.format", "%5$s%n");
	}
	static final Logger LOG = Logger.getLogger ("TreeExampleShots");
	static String dir;

	public static void main (String [] args) {
		dir = args.length > 0 ? args [0] : null;
		Display display = new Display ();
		million (display);
		customDraw (display);
		collapseKeys (display);
		caretSelection (display);
		display.dispose ();
	}

	static void million (Display display) {
		Shell shell = M3TreeMillionNodesExample.open (display);
		Tree tree = find (shell, Tree.class);
		shot (shell, "tree-million-1-open", "roots " + tree.getItemCount () + " top " + text (tree.getTopItem ()));
		long start = System.nanoTime ();
		TreeItem item = tree.getItem (500_000);
		item.setExpanded (true);
		item.getItem (0).setExpanded (true);
		tree.setTopItem (item);
		tree.setSelection (item.getItem (0).getItem (3));
		long micros = (System.nanoTime () - start) / 1000;
		LOG.info (() -> "million: expand and select in the middle took " + micros + " us");
		shot (shell, "tree-million-2-middle-expanded", "top " + text (tree.getTopItem ()) + " count " + item.getItemCount () + " bar max " + tree.getVerticalBar ().getMaximum ()
			+ " selection " + text (tree.getSelection () [0]));
		tree.setTopItem (tree.getItem (999_990));
		shot (shell, "tree-million-3-end", "top " + text (tree.getTopItem ()) + " bar " + tree.getVerticalBar ().getSelection () + "/" + tree.getVerticalBar ().getMaximum ());
		shell.dispose ();
	}

	static void customDraw (Display display) {
		Shell shell = M3TreeCustomDrawExample.open (display);
		Tree tree = find (shell, Tree.class);
		shot (shell, "tree-customdraw-1-open", "item height " + tree.getItemHeight () + " top " + text (tree.getTopItem ()));
		TreeItem item = tree.getItem (500_000);
		item.setExpanded (true);
		tree.setTopItem (item);
		tree.setSelection (item.getItem (2));
		shot (shell, "tree-customdraw-2-middle", "top " + text (tree.getTopItem ()) + " selection " + text (tree.getSelection () [0]));
		M3TreeCustomDrawExample.rowHeight += 8;
		tree.redraw ();
		tree.update ();
		shot (shell, "tree-customdraw-3-taller", "item height " + tree.getItemHeight () + " top " + text (tree.getTopItem ()) + " bar page " + tree.getVerticalBar ().getThumb ());
		shell.dispose ();
		M3TreeCustomDrawExample.rowHeight = 24;
	}

	static void collapseKeys (Display display) {
		Shell shell = M3TreeCollapseKeysExample.open (display);
		Tree tree = find (shell, Tree.class);
		org.eclipse.swt.widgets.List log = find (shell, org.eclipse.swt.widgets.List.class);
		TreeItem root = M3TreeCollapseKeysExample.prepare (tree, log);
		shot (shell, "tree-keys-1-prepared", M3TreeCollapseKeysExample.state (tree, root));
		key (tree, 0x6D /* VK_SUBTRACT */);
		shot (shell, "tree-keys-2-subtract", M3TreeCollapseKeysExample.state (tree, root));
		key (tree, 0x27 /* VK_RIGHT */);
		shot (shell, "tree-keys-3-right", M3TreeCollapseKeysExample.state (tree, root));
		key (tree, 0x25 /* VK_LEFT */);
		shot (shell, "tree-keys-4-left", M3TreeCollapseKeysExample.state (tree, root));
		shell.dispose ();
	}

	static void caretSelection (Display display) {
		Shell shell = M3TreeCaretSelectionExample.open (display);
		Tree tree = find (shell, Tree.class);
		org.eclipse.swt.widgets.List log = find (shell, org.eclipse.swt.widgets.List.class);
		int n = 1;
		for (Map.Entry<String, java.util.function.Consumer<Tree>> entry : M3TreeCaretSelectionExample.CASES.entrySet ()) {
			String line = M3TreeCaretSelectionExample.run (tree, entry.getKey (), entry.getValue (), log);
			shot (shell, "tree-caret-" + n++, line);
		}
		shell.dispose ();
	}

	/* a key the TreeView handles itself, as WM_KEYDOWN and WM_KEYUP (the KeyUp listener of the example logs the state) */
	static void key (Tree tree, int vk) {
		tree.setFocus ();
		org.eclipse.swt.internal.win32.OS.SendMessage (tree.handle, 0x0100, vk, 0);
		org.eclipse.swt.internal.win32.OS.SendMessage (tree.handle, 0x0101, vk, 0xC0000001L);
	}

	static void shot (Shell shell, String name, String state) {
		Display display = shell.getDisplay ();
		long end = System.currentTimeMillis () + 300;
		display.timerExec (300, () -> {});
		while (System.currentTimeMillis () < end) {
			if (!display.readAndDispatch ()) display.sleep ();
		}
		LOG.info (() -> name + ": " + state);
		if (dir == null) return;
		Point size = shell.getSize ();
		Image image = new Image (display, size.x, size.y);
		GC gc = new GC (image);
		shell.print (gc);
		gc.dispose ();
		ImageLoader loader = new ImageLoader ();
		loader.data = new ImageData [] {image.getImageData ()};
		loader.save (dir + "/" + name + ".png", SWT.IMAGE_PNG);
		image.dispose ();
	}

	static <T> T find (Composite parent, Class<T> type) {
		for (Control child : parent.getChildren ()) {
			if (type.isInstance (child)) return type.cast (child);
			if (child instanceof Composite composite && !(child instanceof Tree)) {
				T found = find (composite, type);
				if (found != null) return found;
			}
		}
		return null;
	}

	static String text (TreeItem item) {
		return item == null ? "-" : item.getText ();
	}
}
