function(swt_verify_sha256 path expected label)
  string(LENGTH "${expected}" hash_length)
  if(NOT hash_length EQUAL 64 OR NOT expected MATCHES "^[0-9a-fA-F]+$")
    message(FATAL_ERROR "${label}: supply an exact SHA-256")
  endif()
  if(NOT EXISTS "${path}" OR IS_DIRECTORY "${path}")
    message(FATAL_ERROR "${label}: file is missing: ${path}")
  endif()
  file(SHA256 "${path}" actual)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  string(TOLOWER "${expected}" expected_lower)
  if(NOT actual STREQUAL expected_lower)
    message(FATAL_ERROR "${label}: SHA-256 mismatch")
  endif()
endfunction()

# The application is deliberately a different gate from compiling SWT's native
# library. A host JDK and an SDK do not establish a working iOS JVM closure.
if(SWT_IOS_BUILD_APP)
  set(SWT_IOS_RUNTIME_ROOT "" CACHE PATH "Root of the supplied iOS runtime artifact")
  set(SWT_IOS_RUNTIME_MANIFEST "" CACHE FILEPATH "Pinned runtime manifest; see README.md")
  set(SWT_IOS_RUNTIME_MANIFEST_SHA256 "" CACHE STRING "Expected runtime manifest SHA-256")
  set(SWT_IOS_JDK_SOURCE_REVISION "" CACHE STRING "Expected 40-character JDK source revision")
  set(SWT_IOS_APPLICATION_JAR "" CACHE FILEPATH "Application and its actual dependency closure")
  set(SWT_IOS_MOBILE_JAR_SHA256 "" CACHE STRING "Expected mobile SWT JAR SHA-256")
  set(SWT_IOS_APPLICATION_JAR_SHA256 "" CACHE STRING "Expected application JAR SHA-256")
  set(SWT_IOS_MAIN_CLASS "" CACHE STRING "Existing application class with public static void main(String[])")
  set(SWT_IOS_BUNDLE_ID "org.eclipse.swt.mobile.fixture" CACHE STRING "Application bundle identifier")

  swt_verify_sha256("${SWT_IOS_RUNTIME_MANIFEST}" "${SWT_IOS_RUNTIME_MANIFEST_SHA256}" "runtime manifest")
  swt_verify_sha256("${SWT_IOS_MOBILE_JAR}" "${SWT_IOS_MOBILE_JAR_SHA256}" "mobile SWT JAR")
  swt_verify_sha256("${SWT_IOS_APPLICATION_JAR}" "${SWT_IOS_APPLICATION_JAR_SHA256}" "application JAR")
  string(LENGTH "${SWT_IOS_JDK_SOURCE_REVISION}" revision_length)
  if(NOT revision_length EQUAL 40 OR NOT SWT_IOS_JDK_SOURCE_REVISION MATCHES "^[0-9a-fA-F]+$")
    message(FATAL_ERROR "Supply the actual JDK source commit in SWT_IOS_JDK_SOURCE_REVISION")
  endif()
  if(NOT SWT_IOS_MAIN_CLASS MATCHES "^[A-Za-z_$][A-Za-z0-9_.$]*$")
    message(FATAL_ERROR "Supply a qualified Java application main class")
  endif()
  if(NOT SWT_IOS_BUNDLE_ID MATCHES "^[A-Za-z0-9.-]+$")
    message(FATAL_ERROR "Invalid application bundle identifier")
  endif()
  if(NOT IS_DIRECTORY "${SWT_IOS_RUNTIME_ROOT}")
    message(FATAL_ERROR "SWT_IOS_RUNTIME_ROOT does not exist")
  endif()
  file(REAL_PATH "${SWT_IOS_RUNTIME_ROOT}" runtime_root)
  file(READ "${SWT_IOS_RUNTIME_MANIFEST}" manifest)
  string(JSON schema GET "${manifest}" schemaVersion)
  string(JSON revision GET "${manifest}" sourceRevision)
  string(JSON platform GET "${manifest}" platform)
  string(JSON architecture GET "${manifest}" architecture)
  string(JSON java_home GET "${manifest}" javaHome)
  string(JSON jni_include GET "${manifest}" jniInclude)
  string(JSON jni_platform_include GET "${manifest}" jniPlatformInclude)
  if(NOT schema EQUAL 1 OR NOT revision STREQUAL SWT_IOS_JDK_SOURCE_REVISION)
    message(FATAL_ERROR "Runtime schema or source revision does not match its declared pin")
  endif()
  if(swt_sdk_lower MATCHES "iphonesimulator")
    set(expected_platform iphonesimulator)
  else()
    set(expected_platform iphoneos)
  endif()
  if(NOT platform STREQUAL expected_platform OR NOT architecture STREQUAL CMAKE_OSX_ARCHITECTURES)
    message(FATAL_ERROR "Runtime platform/architecture does not match the selected Xcode target")
  endif()

  function(swt_runtime_path relative output)
    if(relative STREQUAL "" OR IS_ABSOLUTE "${relative}" OR relative MATCHES "(^|/)\\.\\.?(/|$)" OR relative MATCHES "[\\\\;]")
      message(FATAL_ERROR "Runtime manifest path must be a nonescaping relative path: ${relative}")
    endif()
    file(REAL_PATH "${runtime_root}/${relative}" resolved)
    string(FIND "${resolved}" "${runtime_root}/" prefix)
    if(NOT prefix EQUAL 0 OR NOT EXISTS "${resolved}")
      message(FATAL_ERROR "Runtime path escapes its artifact or is missing: ${relative}")
    endif()
    set(${output} "${resolved}" PARENT_SCOPE)
  endfunction()

  string(JSON file_count LENGTH "${manifest}" files)
  if(file_count LESS 1)
    message(FATAL_ERROR "Runtime manifest must inventory every packaged image, header and library file")
  endif()
  math(EXPR file_last "${file_count} - 1")
  set(verified_paths)
  foreach(index RANGE 0 ${file_last})
    string(JSON relative GET "${manifest}" files ${index} path)
    string(JSON expected GET "${manifest}" files ${index} sha256)
    if(relative IN_LIST verified_paths)
      message(FATAL_ERROR "Duplicate runtime inventory entry: ${relative}")
    endif()
    swt_runtime_path("${relative}" resolved)
    swt_verify_sha256("${resolved}" "${expected}" "runtime ${relative}")
    list(APPEND verified_paths "${relative}")
  endforeach()
  foreach(required "${jni_include}/jni.h" "${jni_platform_include}/jni_md.h")
    if(NOT required IN_LIST verified_paths)
      message(FATAL_ERROR "Target JNI header is absent from the verified manifest: ${required}")
    endif()
  endforeach()
  swt_runtime_path("${jni_include}" SWT_IOS_JNI_INCLUDE_DIR)
  swt_runtime_path("${jni_platform_include}" SWT_IOS_JNI_PLATFORM_INCLUDE_DIR)
  swt_runtime_path("${java_home}" runtime_image)
  if(NOT IS_DIRECTORY "${runtime_image}")
    message(FATAL_ERROR "Runtime javaHome must be a directory")
  endif()
  # copy_directory follows directory links. A deployable runtime image must be
  # materialized, so its complete inventory can be checked before packaging.
  file(GLOB_RECURSE runtime_image_entries LIST_DIRECTORIES true "${runtime_image}/*")
  foreach(entry IN LISTS runtime_image_entries)
    if(IS_SYMLINK "${entry}")
      message(FATAL_ERROR "Materialize runtime image symlinks before hashing and packaging: ${entry}")
    endif()
  endforeach()
  file(GLOB_RECURSE runtime_image_files LIST_DIRECTORIES false RELATIVE "${runtime_root}" "${runtime_image}/*")
  if(NOT runtime_image_files)
    message(FATAL_ERROR "Runtime image is empty")
  endif()
  foreach(relative IN LISTS runtime_image_files)
    if(NOT relative IN_LIST verified_paths)
      message(FATAL_ERROR "Unverified file in the packaged runtime image: ${relative}")
    endif()
  endforeach()
  string(JSON library_count LENGTH "${manifest}" libraries)
  if(library_count LESS 1)
    message(FATAL_ERROR "Supply the real static iOS JVM and its complete native library closure")
  endif()
  math(EXPR library_last "${library_count} - 1")
  set(runtime_libraries)
  foreach(index RANGE 0 ${library_last})
    string(JSON relative GET "${manifest}" libraries ${index})
    if(NOT relative IN_LIST verified_paths OR NOT relative MATCHES "\\.a$")
      message(FATAL_ERROR "Runtime library is not an inventoried static archive: ${relative}")
    endif()
    swt_runtime_path("${relative}" library)
    list(APPEND runtime_libraries "${library}")
  endforeach()
  set(runtime_system_libraries)
  foreach(field frameworks systemLibraries)
    string(JSON item_count LENGTH "${manifest}" ${field})
    if(item_count GREATER 0)
      math(EXPR item_last "${item_count} - 1")
      foreach(index RANGE 0 ${item_last})
        string(JSON item GET "${manifest}" ${field} ${index})
        if(NOT item MATCHES "^[A-Za-z0-9_]+$")
          message(FATAL_ERROR "Invalid system library name in runtime manifest")
        endif()
        unset(runtime_system_library CACHE)
        find_library(runtime_system_library NAMES "${item}" REQUIRED)
        list(APPEND runtime_system_libraries "${runtime_system_library}")
      endforeach()
    endif()
  endforeach()
endif()

foreach(required "${SWT_IOS_JNI_INCLUDE_DIR}/jni.h" "${SWT_IOS_JNI_PLATFORM_INCLUDE_DIR}/jni_md.h")
  if(NOT EXISTS "${required}")
    message(FATAL_ERROR "Missing JNI header: ${required}. Supply actual JDK headers, never test stubs.")
  endif()
endforeach()

set(os_source "${SWT_PI_IOS}/org/eclipse/swt/internal/ios/OS.java")

    -cp "${SWT_IOS_MOBILE_JAR}" -d "${CMAKE_CURRENT_BINARY_DIR}/java-classes" -h "${generated}"
    "${os_source}" "${platform_source}"
  DEPENDS "${os_source}" "${platform_source}" "${SWT_IOS_MOBILE_JAR}"
  VERBATIM)
add_custom_target(swt_ios_jni_headers DEPENDS "${jni_header}")

function(swt_ios_compile target)
  target_compile_features(${target} PRIVATE c_std_11)
  target_compile_options(${target} PRIVATE -fobjc-arc -Wall -Wextra -Werror)
  target_include_directories(${target} PRIVATE "${SWT_PI_IOS}/library" "${generated}"
    "${SWT_IOS_JNI_INCLUDE_DIR}" "${SWT_IOS_JNI_PLATFORM_INCLUDE_DIR}")
  set_target_properties(${target} PROPERTIES XCODE_ATTRIBUTE_CLANG_ENABLE_OBJC_ARC YES
    OBJC_STANDARD 11 OBJC_STANDARD_REQUIRED YES)
  add_dependencies(${target} swt_ios_jni_headers)
endfunction()

swt_ios_compile(swt_ios)

    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${runtime_image}" "$<TARGET_BUNDLE_DIR:SWTMobile>/runtime"
    VERBATIM)
